# Platforms without subprocesses run the intro movie player in the game's process: its generated
# code is compiled as one translation unit inside namespace intro (no clashes with the game's).
function(add_intro_in_process target)
  file(GLOB intro_gen RELATIVE ${CMAKE_SOURCE_DIR}/gen_intro ${CMAKE_SOURCE_DIR}/gen_intro/funcs_*.gen.cpp)
  set(src "// Generated by CMakeLists.txt: gen_intro inside namespace intro.\n#include \"runtime/x86/cpu.h\"\n#include <cstddef>\nnamespace intro\n{\n")
  foreach(f ${intro_gen} registry.gen.cpp)
    string(APPEND src "#include \"gen_intro/${f}\"\n")
  endforeach()
  string(APPEND src "const X86Func* funcs(size_t& count)\n{\n    count = g_x86FuncCount;\n    return g_x86Funcs;\n}\n}\n")
  file(WRITE ${CMAKE_BINARY_DIR}/intro_program.cpp "${src}")
  add_library(${target}_intro STATIC ${CMAKE_BINARY_DIR}/intro_program.cpp)
  target_include_directories(${target}_intro PRIVATE ${CMAKE_SOURCE_DIR})
  target_compile_options(${target}_intro PRIVATE -O2 -w -fno-strict-aliasing)
  target_link_libraries(${target} PRIVATE ${target}_intro)
  target_compile_definitions(${target} PRIVATE KP_INTRO_IN_PROCESS=1)
endfunction()

function(guest_program target exe gendir)
  file(GLOB gen ${CMAKE_SOURCE_DIR}/${gendir}/*.gen.cpp)
  add_library(${target}_gen STATIC ${gen})
  target_compile_options(${target}_gen PRIVATE -O2 -w -fno-strict-aliasing)
  if(ANDROID)
    add_library(${target} SHARED ${RUNTIME_SOURCES} ${PLATFORM_SOURCES})
  else()
    add_executable(${target} ${RUNTIME_SOURCES} ${PLATFORM_SOURCES})
  endif()
  target_compile_definitions(${target} PRIVATE KP_EXE_NAME="${exe}")
  target_compile_options(${target} PRIVATE -Wall -Wno-unused-function -Wno-deprecated-declarations -fno-strict-aliasing)
  target_link_libraries(${target} PRIVATE ${target}_gen ${PLATFORM_LIBS})
  if(ANDROID) # NativeActivity's entry point lives in the glue; 16 KB pages
    target_link_options(${target} PRIVATE -u ANativeActivity_onCreate -Wl,-z,max-page-size=16384)
  endif()
endfunction()
